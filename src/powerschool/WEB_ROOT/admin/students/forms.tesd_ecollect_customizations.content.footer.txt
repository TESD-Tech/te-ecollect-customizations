    cursor: pointer;

    margin: 0;

<script>
(function() {
  'use strict';

  // Use direct PS tags for student context and initial value
  const studentDcid = '~(*evaluate mid(~(studentfrn),4,99999))';
  const initialVal = '~([Students.U_Students_Extension]ONLINE_REG_PIPELINE)';
  const updateField = 'ONLINE_REG_PIPELINE';

  const observer = new MutationObserver((mutations, obs) => {
    const anchor = document.querySelector('pss-add-favorite a[id="btnAddRemoveFavorite"]');
    if (anchor && !document.querySelector('.online-reg-toggle-container')) {
      const parent = anchor.closest('pss-add-favorite');
      if (parent && parent.parentNode) {
        const container = document.createElement('div');
        container.className = 'online-reg-toggle-container';
        container.innerHTML = `
          <input type="checkbox" id="ONLINE_REG_PIPELINE">
          <label for="ONLINE_REG_PIPELINE">Show Online Reg Forms</label>
        `;
        
        const cb = container.querySelector('input');
        // PS often returns '1' or '0' string, or empty if null
        cb.checked = (initialVal == '1');

        cb.addEventListener('change', async (e) => {
          const val = e.target.checked ? '1' : '0';
          const endpoint = `/ws/schema/table/U_DEF_EXT_STUDENTS/${studentDcid}`;
          const payload = {
              name: 'STUDENTS',
              tables: {
                  U_DEF_EXT_STUDENTS: {
                      [updateField]: val
                  }
              }
          };

          try {
            const response = await fetch(endpoint, {
                method: 'PUT',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!response.ok) {
                const errorData = await response.json().catch(() => ({ message: response.statusText }));
                throw new Error(errorData.message || `HTTP ${response.status}`);
            }
            
            const result = await response.json();
            console.log('Online Reg Toggle saved via Students table:', result);
          } catch (error) {
             console.error('Error saving Online Reg Toggle:', error);
             alert('Error saving change: ' + error.message);
             // Revert checkbox
             e.target.checked = !e.target.checked;
          }
        });

        parent.parentNode.insertBefore(container, parent.nextSibling);
        obs.disconnect();
      }
    }
  });

  observer.observe(document.body, { childList: true, subtree: true });
})();
</script>
